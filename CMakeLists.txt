    PREFIX "/"

# Icons are generated by tools/make_icons.py.
if(WIN32)
    # Executable icon and version info (shown in Explorer, Task Manager and notifications).
    set(APP_ICO ${CMAKE_CURRENT_SOURCE_DIR}/assets/icon/app.ico)
    configure_file(assets/windows/app.rc.in ${CMAKE_CURRENT_BINARY_DIR}/app.rc @ONLY)
    target_sources(SilentSoftphoneFree PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/app.rc)
elseif(APPLE)
    set(APP_ICNS ${CMAKE_CURRENT_SOURCE_DIR}/assets/icon/app.icns)
    set_source_files_properties(${APP_ICNS} PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
    target_sources(SilentSoftphoneFree PRIVATE ${APP_ICNS})
endif()

target_include_directories(SilentSoftphoneFree PRIVATE src)

# PJSIP backend: build PJSIP with tools/build_pjsip.sh and pass -DPj_DIR=<prefix>/lib/cmake/Pj.
# Without it the app builds with the mock SIP backend only.
option(SILENTSOFTPHONE_WITH_PJSIP "Build the PJSIP SIP backend" ON)
option(SILENTSOFTPHONE_REQUIRE_PJSIP "Fail the configure step if PJSIP is missing (CI)" OFF)
if(SILENTSOFTPHONE_WITH_PJSIP)
    find_package(Pj CONFIG QUIET)
    if(Pj_FOUND)
        message(STATUS "PJSIP ${Pj_VERSION}: SIP backend enabled")
        target_sources(SilentSoftphoneFree PRIVATE
            src/sip/PjsipService.h
            src/sip/PjsipService.cpp
        )
        target_compile_definitions(SilentSoftphoneFree PRIVATE SILENTSOFTPHONE_HAS_PJSIP)
        if(WIN32)
            target_compile_definitions(SilentSoftphoneFree PRIVATE NOMINMAX)
        endif()
        target_link_libraries(SilentSoftphoneFree PRIVATE Pj::pjsua2)
    elseif(SILENTSOFTPHONE_REQUIRE_PJSIP)
        message(FATAL_ERROR "PJSIP not found (set Pj_DIR)")
    else()
        message(WARNING "PJSIP not found (set Pj_DIR); building with the mock SIP backend only")
    endif()
endif()

target_compile_definitions(SilentSoftphoneFree PRIVATE
    SILENTSOFTPHONE_VERSION="${PROJECT_VERSION}"
)

set_target_properties(SilentSoftphoneFree PROPERTIES
    WIN32_EXECUTABLE TRUE
    MACOSX_BUNDLE TRUE
    MACOSX_BUNDLE_BUNDLE_NAME "Silent Softphone Free"
    MACOSX_BUNDLE_GUI_IDENTIFIER io.github.silentcasting.silentsoftphonefree
    MACOSX_BUNDLE_ICON_FILE app.icns
    MACOSX_BUNDLE_INFO_PLIST ${CMAKE_CURRENT_SOURCE_DIR}/assets/macos/Info.plist.in
    MACOSX_BUNDLE_BUNDLE_VERSION ${PROJECT_VERSION}
    MACOSX_BUNDLE_SHORT_VERSION_STRING ${PROJECT_VERSION_MAJOR}.${PROJECT_VERSION_MINOR}
)

# macOS users see the bundle name: "Silent Softphone Free.app".
if(APPLE)
    set_target_properties(SilentSoftphoneFree PROPERTIES OUTPUT_NAME "Silent Softphone Free")
endif()

target_link_libraries(SilentSoftphoneFree PRIVATE
    Qt6::Quick
    Qt6::QuickControls2
    Qt6::Widgets
)

install(TARGETS SilentSoftphoneFree
    BUNDLE DESTINATION .
    RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR}
)
